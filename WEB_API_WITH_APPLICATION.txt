/* Web API Expression - Routes to appropriate plugin function based on object type */

a!localVariables(
  local!objectType: http!request.queryParameters.type,
  local!objectName: http!request.queryParameters.name,
  local!credentialKey: cons!CONS_AI_CREDENTIAL_KEY_PLUGIN,

  local!result: if(
    lower(local!objectType) = "application",
    fn!explainApplication(
      applicationName: local!objectName,
      externalSystemKey: local!credentialKey
    ),
    fn!explainAppianObject(
      objectType: local!objectType,
      objectName: local!objectName,
      externalSystemKey: local!credentialKey
    )
  ),

  a!httpResponse(
    statusCode: 200,
    headers: { a!httpHeader(name: "Content-Type", value: "application/json") },
    body: a!toJson({ result: local!result })
  )
)
